# 配置驱动训练教程

> 把超参数抽离到 YAML 文件中，方便切换算法和调参。

---

## 1. 为什么用配置驱动？

| 不使用 | 使用配置驱动 |
|--------|-------------|
| 改参数要进代码改 | 改 YAML 文件即可 |
| 换算法要改 import | 改一行 `algorithm: PPO` |
| 参数分散难追踪 | 参数集中易管理 |
| 同时跑多个实验困难 | 多份 YAML 轻松对比 |

## 5. train.py - 训练脚本

```python
"""
配置驱动训练脚本
使用方法：
    python workspace/train.py
    python workspace/train.py algorithm=SAC
"""

import argparse
import yaml
from pathlib import Path

from metadrive import MetaDriveEnv
from metadrive.component.sensors.rgb_camera import RGBCamera
from metadrive.component.map.base_map import BaseMap
from metadrive.component.map.pg_map import MapGenerateMethod

from stable_baselines3 import PPO, SAC


# ==================== 加载配置 ====================

def load_config(algorithm="PPO"):
    """加载配置，合并 base + 算法专用配置"""
    base = yaml.safe_load(open("configs/base.yaml"))

    algo_file = f"configs/{algorithm.lower()}.yaml"
    if Path(algo_file).exists():
        algo_cfg = yaml.safe_load(open(algo_file))
    else:
        print(f"警告：{algo_file} 不存在，使用默认参数")
        algo_cfg = {}

    # 合并配置
    return {**base, **algo_cfg}


# ==================== 创建环境 ====================

def make_env(config):
    """根据配置创建 MetaDrive 环境"""
    map_config = {
        BaseMap.GENERATE_TYPE: MapGenerateMethod.BIG_BLOCK_NUM,
        BaseMap.GENERATE_CONFIG: config["map_segments"],
        BaseMap.LANE_WIDTH: config["lane_width"],
        BaseMap.LANE_NUM: config["lane_num"],
    }

    w, h = config.get("camera_size", [320, 180])

    env_config = dict(
        use_render=config.get("use_render", False),
        manual_control=False,
        traffic_density=config["traffic_density"],
        num_scenarios=config["num_scenarios"],
        random_agent_model=False,
        on_continuous_line_done=config["on_continuous_line_done"],
        out_of_route_done=config["out_of_route_done"],
        image_observation=config["image_observation"],
        sensors=dict(rgb_camera=(RGBCamera, w, h)),
        norm_pixel=config["norm_pixel"],
        vehicle_config=dict(
            show_lidar=False,
            show_navi_mark=False,
            show_line_to_navi_mark=False,
        ),
        map_config=map_config,
    )
    return MetaDriveEnv(env_config)


# ==================== 主程序 ====================

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--algorithm", type=str, default="PPO",
                        help="选择算法：PPO 或 SAC")
    args = parser.parse_args()

    # 加载配置
    config = load_config(args.algorithm)
    print(f"使用算法: {args.algorithm}")
    print(f"训练步数: {config['total_timesteps']}")

    # 创建环境
    env = make_env(config)

    # 创建模型
    if args.algorithm == "PPO":
        from stable_baselines3 import PPO
        model = PPO(
            "MultiInputPolicy",
            env,
            learning_rate=config["learning_rate"],
            n_steps=config["n_steps"],
            batch_size=config["batch_size"],
            n_epochs=config["n_epochs"],
            gamma=config["gamma"],
            gae_lambda=config["gae_lambda"],
            clip_range=config["clip_range"],
            tensorboard_log=config["tensorboard_log"],
            verbose=1,
        )
    elif args.algorithm == "SAC":
        from stable_baselines3 import SAC
        model = SAC(
            "MultiInputPolicy",
            env,
            learning_rate=config["learning_rate"],
            batch_size=config["batch_size"],
            gamma=config["gamma"],
            tensorboard_log=config["tensorboard_log"],
            verbose=1,
        )

    # 训练
    print(f"开始训练，共 {config['total_timesteps']} 步")
    model.learn(total_timesteps=config["total_timesteps"], progress_bar=True)

    # 保存
    save_path = config["save_path"]
    model.save(save_path)
    print(f"模型已保存: {save_path}")

    env.close()
```

## 6. 运行方式

```bash
# 默认 PPO

python workspace/train.py --algorithm PPO
```

## 7. 快速开始（不用新建文件）

如果觉得创建文件夹麻烦，可以先用**简单变量**：

```python
# 最简单的配置驱动
ALGORITHM = "PPO"  # 改这里就能换算法
LEARNING_RATE = 0.0003
TOTAL_STEPS = 5000  # 先小量测试

if ALGORITHM == "PPO":
    from stable_baselines3 import PPO
    model = PPO("MultiInputPolicy", env, learning_rate=LEARNING_RATE)
elif ALGORITHM == "SAC":
    from stable_baselines3 import SAC
    model = SAC("MultiInputPolicy", env, learning_rate=LEARNING_RATE)
```

## 总结

| 阶段 | 推荐做法 |
|------|---------|
| 刚开始 | 用简单变量（改 `ALGORITHM = "SAC"`） |
| 开始调参 | 用 YAML 分文件管理 |
| 写论文 | 用 Hydra + 多份 YAML 对比 |

先跑通训练，理解流程，之后再考虑要不要上 YAML 配置管理。
